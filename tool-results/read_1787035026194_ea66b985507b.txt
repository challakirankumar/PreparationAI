   306→   306→   306→function daysToExam(examDate?: string): number {
   307→   307→   307→  if (!examDate) return 0;
   308→   308→   308→  const today = new Date();
   309→   309→   309→  today.setHours(0, 0, 0, 0);
   310→   310→   310→  const target = new Date(examDate);
   311→   311→   311→  target.setHours(0, 0, 0, 0);
   312→   312→   312→  return Math.max(0, Math.ceil((target.getTime() - today.getTime()) / 86400000));
   313→   313→   313→}
