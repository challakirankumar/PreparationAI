    16→let _idCounter = 0;
    17→
    18→export function uniqueId(): string {
    19→  _idCounter += 1;
    20→  return `q${Date.now().toString(36)}${_idCounter.toString(36)}${Math.random().toString(36).slice(2, 6)}`;
    21→}
    22→
    23→export function randInt(min: number, max: number): number {
    24→  return Math.floor(Math.random() * (max - min + 1)) + min;
    25→}
    26→
    27→export function randFloat(min: number, max: number, decimals = 2): number {
    28→  const v = Math.random() * (max - min) + min;
    29→  const f = Math.pow(10, decimals);
    30→  return Math.round(v * f) / f;
    31→}
    32→
    33→export function pick<T>(arr: T[]): T {
    34→  return arr[Math.floor(Math.random() * arr.length)];
    35→}
    36→
    37→export function shuffle<T>(arr: T[]): T[] {
    38→  const a = [...arr];
    39→  for (let i = a.length - 1; i > 0; i--) {
    40→    const j = Math.floor(Math.random() * (i + 1));
    41→    const tmp = a[i];
    42→    a[i] = a[j];
    43→    a[j] = tmp;
    44→  }
    45→  return a;
    46→}

    48→// ============================================================================
    49→// NUMBER FORMATTING (prevents floating-point noise like 7.380000000000001)
    50→// ============================================================================
    51→
    52→export function fmtNum(n: number): string {
    53→  if (!isFinite(n)) return '0';
    54→  const r = Math.round(n * 100) / 100;
    55→  if (Number.isInteger(r)) return String(r);
    56→  return r.toFixed(2).replace(/0+$/, '').replace(/\.$/, '');
    57→}

    63→export function signature(text: string): string {
    64→  return text
    65→    .toLowerCase()
    66→    .replace(/\d+/g, '#')
    67→    .replace(/\s+/g, ' ')
    68→    .trim()
    69→    .slice(0, 80);
    70→}

    72→// ============================================================================
    73→// OPTION BUILDERS

    75→
    76→export function numericOptions(
    77→  answer: number,
    78→  unit = '',
    79→): { options: string[]; correct: number } {
    80→  const set = new Set<number>([answer]);
    81→  const baseDeltas = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.5, 1, 2, 5, 10];
    82→  let safety = 0;
    83→  while (set.size < 4 && safety < 200) {
    84→    safety += 1;
    85→    const ratio = pick(baseDeltas);
    86→    const sign = Math.random() < 0.5 ? -1 : 1;
    87→    const candidate =
    88→      Math.round(answer * (1 + sign * ratio) * 100) / 100;
    89→    if (candidate !== answer && candidate >= 0 && isFinite(candidate)) {
    90→      set.add(candidate);
    91→    }
    92→  }
    93→  // Fallback if we couldn't generate enough distinct distractors
    94→  let extra = 1;
    95→  while (set.size < 4) {
    96→    set.add(answer + extra);
    97→    extra += 1;
    98→  }
    99→  const arr = shuffle([...set]);
   100→  const correct = arr.indexOf(answer);
   101→  const options = arr.map((o) => `${fmtNum(o)}${unit ? ' ' + unit : ''}`);
   102→  return { options, correct };
   103→}

   105→export function mcqFromOptions(
   106→  text: string,
   107→  correct: string,
   108→  distractors: string[],
   109→  meta: QuestionMeta,
   110→): Question {
   111→  const seen = new Set<string>([correct.toLowerCase().trim()]);
   112→  const cleanD: string[] = [];
   113→  for (const d of distractors) {
   114→    const k = String(d).toLowerCase().trim();
   115→    if (!seen.has(k)) {
   116→      seen.add(k);
   117→      cleanD.push(d);
   118→    }
   119→    if (cleanD.length >= 3) break;
   120→  }
   121→  while (cleanD.length < 3) cleanD.push('None of the above');
   122→  const all = shuffle([correct, ...cleanD]);
   123→  const correctIdx = all.indexOf(correct);
   124→  return {
   125→    id: uniqueId(),
   126→    type: 'mcq',
   127→    subject: meta.subject,
   128→    topic: meta.topic,
   129→    difficulty: meta.difficulty,
   130→    text,
   131→    options: all,
   132→    correctOptions: [correctIdx],
   133→    marks: meta.marks,
   134→    negativeMarks: meta.negativeMarks,
   135→  };
   136→}

   138→// Helper for numeric MCQs (reduces boilerplate across parameterised generators)
   139→function numericMcq(
   140→  text: string,
   141→  answer: number,
   142→  unit: string,
   143→  meta: QuestionMeta,
   144→): Question {
   145→  const { options, correct } = numericOptions(answer, unit);
   146→  return {
   147→    id: uniqueId(),
   148→    type: 'mcq',
   149→    subject: meta.subject,
   150→    topic: meta.topic,
   151→    difficulty: meta.difficulty,
   152→    text,
   153→    options,
   154→    correctOptions: [correct],
   155→    marks: meta.marks,
   156→    negativeMarks: meta.negativeMarks,
   157→    unit,
   158→  };
   159→}

   165→export interface BankQuestion {
   166→  topic: string;
   167→  text: string;
   168→  // opts[0] is correct; remaining are distractors.
   169→  // Defensive: opts may contain nested arrays (legacy data bug), so we flatten.
   170→  opts: unknown[];
   171→  explanation?: string;
   172→}
   173→
   174→// Compact bank-question constructor for static banks below.
   175→function BQ(
   176→  topic: string,
   177→  text: string,
   178→  correct: string,
   179→  ...distractors: string[]
   180→): BankQuestion {
   181→  return { topic, text, opts: [correct, ...distractors] };
   182→}
   183→
   184→export function fromBank(
   185→  bank: BankQuestion[],
   186→  meta: QuestionMeta,
   187→  usedTexts?: Set<string>,
   188→): Question {
   189→  const shuffled = shuffle(bank);
   190→  for (const bq of shuffled) {
   191→    if (usedTexts && usedTexts.has(bq.text)) continue;
   192→
   193→    // Flatten nested arrays defensively (legacy opts shape:
   194→    // ['correct', ['d1','d2','d3'] as unknown as string[]])
   195→    const flat: string[] = [];
   196→    for (const o of bq.opts) {
   197→      if (Array.isArray(o)) {
   198→        for (const inner of o) flat.push(String(inner));
   199→      } else {
   200→        flat.push(String(o));
   201→      }
   202→    }
   203→    if (flat.length < 2) continue;
   204→
   205→    const correct = flat[0];
   206→    const distractors = flat.slice(1);
   207→
   208→    // Dedupe options within question (case-insensitive)
   209→    const seen = new Set<string>([correct.toLowerCase().trim()]);
   210→    const deduped: string[] = [correct];
   211→    for (const d of distractors) {
   212→      const k = String(d).toLowerCase().trim();
   213→      if (!seen.has(k)) {
   214→        seen.add(k);
   215→        deduped.push(d);
   216→      }
   217→      if (deduped.length >= 4) break;
   218→    }
   219→    if (deduped.length < 2) continue;
   220→
   221→    // Shuffle so the correct answer isn't always at index 0
   222→    const opts = shuffle(deduped);
   223→    const correctIdx = opts.indexOf(correct);
   224→
   225→    if (usedTexts) usedTexts.add(bq.text);
   226→
   227→    return {
   228→      id: uniqueId(),
   229→      type: 'mcq',
   230→      subject: meta.subject,
   231→      topic: meta.topic,
   232→      difficulty: meta.difficulty,
   233→      text: bq.text,
   234→      options: opts,
   235→      correctOptions: [correctIdx],
   236→      marks: meta.marks,
   237→      negativeMarks: meta.negativeMarks,
   238→    };
   239→  }
   240→
   241→  // Fallback if bank is exhausted
   242→  return mcqFromOptions(
   243→    `Which statement about ${meta.topic} is correct?`,
   244→    'It is a fundamental concept in the subject.',
   245→    [
   246→      'It is unrelated to the subject.',
   247→      'It has no practical application.',
   248→      'It only applies in rare circumstances.',
   249→    ],
   250→    meta,
   251→  );
   252→}

   254→// ============================================================================

   258→export function pickTopic(
   259→  topicWeights: { topic: string; weight: number }[],
   260→  preferTopics?: string[],
   261→): string {
   262→  let pool = topicWeights;
   263→  if (preferTopics && preferTopics.length > 0) {
   264→    const filtered = topicWeights.filter((tw) => preferTopics.includes(tw.topic));
   265→    if (filtered.length > 0) pool = filtered;
   266→  }
   267→  const total = pool.reduce((s, tw) => s + tw.weight, 0);
   268→  let r = Math.random() * total;
   269→  for (const tw of pool) {
   270→    r -= tw.weight;
   271→    if (r <= 0) return tw.topic;
   272→  }
   273→  return pool[pool.length - 1].topic;
   274→}

   276→function leastUsedTopics(
   277→  topicUsage: Map<string, number>,
   278→  topicWeights: { topic: string; weight: number }[],
   279→  n: number,
   280→): string[] {
   281→  const sorted = [...topicWeights].sort(
   282→    (a, b) =>
   283→      (topicUsage.get(a.topic) ?? 0) - (topicUsage.get(b.topic) ?? 0),
   284→  );
   285→  return sorted.slice(0, Math.min(n, sorted.length)).map((tw) => tw.topic);
   286→}

   288→// ============================================================================
   289→// SECTION + EXAM GENERATION
   290→// ============================================================================
   291→
   292→export function generateForSection(
   293→  subject: string,
   294→  topicWeights: { topic: string; weight: number }[],
   295→  count: number,
   296→  section: ExamSection,
   297→  crossAttemptSeen?: Set<string>,
   298→): Question[] {
   299→  const questions: Question[] = [];
   300→  const usedTexts = new Set<string>();
   301→  const usedSignatures = new Set<string>(crossAttemptSeen ?? []);
   302→  const topicUsage = new Map<string, number>();
   303→  for (const tw of topicWeights) topicUsage.set(tw.topic, 0);
   304→
   305→  for (let i = 0; i < count; i++) {
   306→    const ratio = i / Math.max(count, 1);
   307→    const difficulty: 'easy' | 'medium' | 'hard' =
   308→      ratio < 0.3 ? 'easy' : ratio < 0.8 ? 'medium' : 'hard';
   309→
   310→    let generated: Question | null = null;
   311→    let currentTopic = pickTopic(
   312→      topicWeights,
   313→      leastUsedTopics(topicUsage, topicWeights, 2),
   314→    );
   315→
   316→    for (let attempt = 0; attempt < 10; attempt++) {
   317→      // Topic-fallback after 5 failed retries: try a different topic
   318→      if (attempt >= 5) {
   319→        const others = topicWeights.filter((tw) => tw.topic !== currentTopic);
   320→        if (others.length > 0) {
   321→          currentTopic = pickTopic(
   322→            others,
   323→            leastUsedTopics(topicUsage, others, 2),
   324→          );
   325→        }
   326→      }
   327→
   328→      const meta: QuestionMeta = {
   329→        subject,
   330→        topic: currentTopic,
   331→        difficulty,
   332→        marks: section.marksPerQuestion,
   333→        negativeMarks: section.negativeMarks,
   334→      };
   335→
   336→      const gen = GENERATORS[`${subject}|${currentTopic}`];
   337→      if (!gen) continue;
   338→
   339→      let q: Question;
   340→      try {
   341→        q = gen(meta, usedTexts);
   342→      } catch {
   343→        continue;
   344→      }
   345→
   346→      const sig = signature(q.text);
   347→      if (usedSignatures.has(sig) || usedTexts.has(q.text)) continue;
   348→
   349→      generated = q;
   350→      usedSignatures.add(sig);
   351→      usedTexts.add(q.text);
   352→      topicUsage.set(currentTopic, (topicUsage.get(currentTopic) ?? 0) + 1);
   353→      break;
   354→    }
   355→
   356→    if (!generated) {
   357→      // Last-resort fallback: force-generate any question (signature may collide)
   358→      const topic = pickTopic(topicWeights);
   359→      const meta: QuestionMeta = {
   360→        subject,
   361→        topic,
   362→        difficulty,
   363→        marks: section.marksPerQuestion,
   364→        negativeMarks: section.negativeMarks,
   365→      };
   366→      const gen = GENERATORS[`${subject}|${topic}`];
   367→      if (gen) {
   368→        try {
   369→          const q = gen(meta, usedTexts);
   370→          usedTexts.add(q.text);
   371→          usedSignatures.add(signature(q.text));
   372→          generated = q;
   373→        } catch {
   374→          // ignore
   375→        }
   376→      }
   377→    }
   378→
   379→    if (generated) questions.push(generated);
   380→  }
   381→
   382→  return questions;
   383→}

   385→export function generateExam(
   386→  pattern: ExamPattern,
   387→  seenSignatures?: Set<string>,
   388→): {
   389→  questions: Question[];
   390→  sections: { name: string; subject: string; questionIds: string[] }[];
   391→} {
   392→  const allQuestions: Question[] = [];
   393→  const sections: { name: string; subject: string; questionIds: string[] }[] =
   394→    [];
   395→  // Seed internal accumulator with caller-provided signatures; this set
   396→  // accumulates every signature generated during this run so that subsequent
   397→  // sections (and subsequent attempts, via the mutation below) avoid repeats.
   398→  const crossAttemptSeen = new Set<string>(seenSignatures ?? []);
   399→
   400→  for (const section of pattern.sections) {
   401→    const syllabus = pattern.syllabus.find((s) => s.subject === section.subject);
   402→    const topicWeights = syllabus?.topics ?? [];
   403→    const sectionQs = generateForSection(
   404→      section.subject,
   405→      topicWeights,
   406→      section.questionCount,
   407→      section,
   408→      crossAttemptSeen,
   409→    );
   410→    // Propagate this section's signatures back into the accumulator so the
   411→    // next section (and the caller, via the mutation below) sees them.
   412→    for (const q of sectionQs) crossAttemptSeen.add(signature(q.text));
   413→    sections.push({
   414→      name: section.name,
   415→      subject: section.subject,
   416→      questionIds: sectionQs.map((q) => q.id),
   417→    });
   418→    allQuestions.push(...sectionQs);
   419→  }
   420→
   421→  // Propagate newly-seen signatures back to the caller's set so that the
   422→  // same Set can be passed to generateExam again for cross-attempt dedup.
   423→  if (seenSignatures) {
   424→    for (const sig of crossAttemptSeen) seenSignatures.add(sig);
   425→  }
   426→
   427→  return { questions: allQuestions, sections };
   428→}

   434→const G = 9.8; // m/s^2
   435→
   436→export function physicsKinematics(
   437→  m: QuestionMeta,
   438→  _usedTexts?: Set<string>,
   439→): Question {
   440→  const v = randInt(0, 6);
   441→  switch (v) {
   442→    case 0: {
   443→      const d = randInt(50, 500);
   444→      const t = randInt(5, 30);
   445→      const ans = d / t;
   446→      return numericMcq(
   447→        `A car travels ${d} m in ${t} s on a straight road. What is its average velocity?`,
   448→        ans,
   449→        'm/s',
   450→        m,
   451→      );
   452→    }
   453→    case 1: {
   454→      const vel = randInt(5, 30);
   455→      const t = randInt(5, 25);
   456→      const d = vel * t;
   457→      return numericMcq(
   458→        `An object moves at a constant velocity of ${vel} m/s for ${t} s. How far does it travel?`,
   459→        d,
   460→        'm',
   461→        m,
   462→      );
   463→    }
   464→    case 2: {
   465→      const t = randInt(2, 8);
   466→      const h = 0.5 * G * t * t;
   467→      return numericMcq(
   468→        `A ball is dropped from rest. How far does it fall in ${t} s (g = 9.8 m/s²)?`,
   469→        h,
   470→        'm',
   471→        m,
   472→      );
   473→    }
   474→    case 3: {
   475→      const u = randInt(10, 40);
   476→      const hMax = (u * u) / (2 * G);
   477→      return numericMcq(
   478→        `A ball is thrown vertically upward with an initial velocity of ${u} m/s. What is the maximum height it reaches (g = 9.8 m/s²)?`,
   479→        hMax,
   480→        'm',
   481→        m,
   482→      );
   483→    }
   484→    case 4: {
   485→      const u = randInt(20, 60);
   486→      const a = randInt(2, 8);
   487→      const t = randInt(2, 6);
   488→      const finalV = u - a * t;
   489→      return numericMcq(
   490→        `A vehicle moving at ${u} m/s decelerates at ${a} m/s² for ${t} s. What is its final velocity?`,
   491→        finalV,
   492→        'm/s',
   493→        m,
   494→      );
   495→    }
   496→    case 5: {
   497→      const u = randInt(5, 25);
   498→      const v = randInt(30, 60);
   499→      const avg = (u + v) / 2;
   500→      return numericMcq(
   501→        `An object accelerates uniformly from ${u} m/s to ${v} m/s. What is its average velocity?`,
   502→        avg,
   503→        'm/s',
   504→        m,
   505→      );
   506→    }
   507→    default: {
   508→      const u = randInt(10, 30);
   509→      const v = randInt(35, 60);
   510→      const t = randInt(3, 10);
   511→      const a = (v - u) / t;
   512→      return numericMcq(
   513→        `A body accelerates from ${u} m/s to ${v} m/s in ${t} s. What is its acceleration?`,
   514→        a,
   515→        'm/s²',
   516→        m,
   517→      );
   518→    }
   519→  }
   520→}

   584→    case 5: {

   587→      const T = mass * (G + a);
   588→      return numericMcq(
   589→        `A ${mass} kg mass is lifted upward with acceleration ${a} m/s². What is the tension in the rope (g = 9.8 m/s²)?`,
   590→        T,
   591→        'N',
   592→        m,
   593→      );
   594→    }
   595→    default: {
   596→      const mass = randInt(2, 10);
   597→      const a = randInt(1, 6);
   598→      const Fnet = mass * a;
   599→      return numericMcq(
   600→        `What net force is required to give a ${mass} kg object an acceleration of ${a} m/s²?`,
   601→        Fnet,
   602→        'N',
   603→        m,
   604→      );
   605→    }
   606→  }

   609→export function physicsWorkEnergy(
   610→  m: QuestionMeta,
   611→  _usedTexts?: Set<string>,
   612→): Question {
   613→  const v = randInt(0, 6);
   614→  switch (v) {
   615→    case 0: {
   616→      const mass = randInt(1, 20);
   617→      const vel = randInt(2, 20);
   618→      const ke = 0.5 * mass * vel * vel;
   619→      return numericMcq(
   620→        `What is the kinetic energy of a ${mass} kg object moving at ${vel} m/s?`,
   621→        ke,
   622→        'J',
   623→        m,
   624→      );
   625→    }
   626→    case 1: {
   627→      const mass = randInt(1, 20);
   628→      const h = randInt(2, 30);
   629→      const pe = mass * G * h;
   630→      return numericMcq(
   631→        `What is the potential energy of a ${mass} kg object at a height of ${h} m (g = 9.8 m/s²)?`,
   632→        pe,
   633→        'J',
   634→        m,
