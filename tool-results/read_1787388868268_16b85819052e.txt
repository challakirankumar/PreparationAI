   140→      toast({
   141→        title: 'Image too large',
   142→        description: 'Please pick an image under 2 MB.',
   143→        variant: 'destructive',
   144→      });

   149→      const base64 = reader.result as string;
   150→      updateProfile({ avatar: base64 });
   151→      toast({ title: 'Avatar updated' });
   152→    };

   154→      toast({ title: 'Failed to read file', variant: 'destructive' });
   155→    reader.readAsDataURL(file);
   156→    // Reset the input so picking the same file again re-fires onChange
   157→    e.target.value = '';

   183→  function changePassword() {
   184→    if (!user) return;
   185→    const email = user.email.toLowerCase().trim();
   186→    const saved = registeredUsers[email];
   187→    const storedPw = saved?.password || '';
   188→    if (currentPw !== storedPw) {
   189→      toast({ title: 'Current password is incorrect', variant: 'destructive' });
   190→      return;
   191→    }
   192→    if (newPw.length < 6) {
   193→      toast({
   194→        title: 'New password too short',
   195→        description: 'At least 6 characters.',
   196→        variant: 'destructive',
   197→      });

   211→        },
   212→      },
   213→    }));

   216→    setConfirmPw('');

   218→      title: 'Password updated',

   289→  async function saveAndAnalyze() {
   290→    if (!user) return;
   291→    if (!examName.trim()) {
   292→      toast({ title: 'Enter exam name', variant: 'destructive' });
   293→      return;
   294→    }
   295→    const validSubjects = subjects.filter(
   296→      (s) => s.name.trim() && s.maxMarks.trim() !== ''
   297→    );
   298→    if (validSubjects.length === 0) {
   299→      toast({
   300→        title: 'Add at least one subject with marks',
   301→        variant: 'destructive',
   302→      });
   303→      return;
   304→    }
   305→
   306→    const record: AcademicRecord = {
   307→      id: uidGen(),
   308→      examName: examName.trim(),
   309→      institution: institution.trim() || undefined,
   310→      date: examDate,
   311→      uploadedAt: new Date().toISOString(),
   312→      subjects: validSubjects.map((s) => ({
   313→        name: s.name.trim(),
   314→        marks: parseFloat(s.marks) || 0,
   315→        maxMarks: parseFloat(s.maxMarks) || 0,
   316→        grade: s.grade.trim() || undefined,
   317→      })),
   318→      totalMarks: totals.totalMarks,
   319→      maxMarks: totals.maxMarks,
   320→      percentage: totals.percentage,
   321→    };
   322→
   323→    addAcademicRecord(record);
   324→    setUploadOpen(false);
   325→    setAnalyzing(true);
   326→    toast({
   327→      title: 'Record saved',
   328→      description: 'Running AI analysis…',
   329→    });
   330→
   331→    try {
   332→      const res = await fetch('/api/analyze-academic', {
   333→        method: 'POST',
   334→        headers: { 'Content-Type': 'application/json' },
   335→        body: JSON.stringify({
   336→          record,
   337→          examGoal: user.examGoal,
   338→          userName: user.name,
   339→        }),
   340→      });
   341→      const data = await res.json();
   342→      const analysis: AcademicAnalysis | undefined = data?.analysis;
   343→      if (!analysis) {
   344→        toast({
   345→          title: 'Analysis failed',
   346→          description: data?.error || 'No analysis returned.',
   347→          variant: 'destructive',
   348→        });
   349→        return;
   350→      }
   351→      updateAcademicRecord(record.id, { aiAnalysis: analysis });
   352→      if (data.fallback) {
   353→        toast({
   354→          title: 'Analysis ready (offline mode)',
   355→          description: 'Used deterministic fallback. Try again later for full AI insights.',
   356→        });
   357→      } else {
   358→        toast({
   359→          title: 'AI analysis complete',
   360→          description: 'Tap “View AI Analysis” on the record to read it.',
   361→        });
   362→      }
   363→    } catch (e) {
   364→      toast({
   365→        title: 'Analysis failed',
   366→        description: (e as Error).message,
   367→        variant: 'destructive',
   368→      });
   369→    } finally {
   370→      setAnalyzing(false);
   371→      resetUploadForm();
   372→    }
   373→  }

  1022→function AcademicRecordCard({
  1023→  record,
  1024→  onViewAnalysis,

  1034→    accent === 'blue'
  1035→      ? 'bg-blue-500'

  1037→      ? 'bg-amber-500'

  1040→    1,

  1046→  return (

  1069→                : accent === 'amber'
  1070→                ? 'bg-amber-100 text-amber-700 border-amber-200'
  1071→                : 'bg-rose-100 text-rose-700 border-rose-200'
