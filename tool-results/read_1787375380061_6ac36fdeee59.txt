    30→function gradeFor(pct: number): string {
    31→  if (pct >= 90) return 'A+';
    32→  if (pct >= 80) return 'A';
    33→  if (pct >= 70) return 'B+';
    34→  if (pct >= 60) return 'B';
    35→  if (pct >= 50) return 'C';
    36→  if (pct >= 40) return 'D';
    37→  return 'E';
    38→}
    39→
    40→function fmtDuration(sec: number): string {
    41→  if (sec < 60) return `${sec}s`;
    42→  const m = Math.floor(sec / 60);
    43→  const s = sec % 60;
    44→  if (m < 60) return s > 0 ? `${m}m ${s}s` : `${m}m`;
    45→  const h = Math.floor(m / 60);
    46→  const mm = m % 60;
    47→  return mm > 0 ? `${h}h ${mm}m` : `${h}h`;
    48→}

   304→// ----------------- Topics tab -----------------
   305→function TopicsTab({ attempt }: { attempt: ExamAttempt }) {
   306→  const topics = attempt.topicScores;
   307→  function colorFor(acc: number): string {
   308→    if (acc >= 75) return 'bg-blue-500 text-white';
   309→    if (acc >= 60) return 'bg-blue-300 text-blue-900';
   310→    if (acc >= 45) return 'bg-amber-300 text-amber-900';
   311→    if (acc >= 30) return 'bg-amber-500 text-white';
   312→    return 'bg-rose-500 text-white';
   313→  }
   314→  return (
   315→    <Card className="border-stone-200">
   316→      <CardHeader>
   317→        <CardTitle className="text-base flex items-center gap-2">
   318→          <Flame className="h-4 w-4 text-blue-600" /> Topic-wise accuracy heatmap
   319→        </CardTitle>
   320→        <CardDescription>Color-coded by accuracy — hover for details</CardDescription>
   321→      </CardHeader>
   322→      <CardContent>
   323→        {topics.length === 0 ? (
   324→          <p className="text-sm text-muted-foreground">No topic breakdown available.</p>
   325→        ) : (
   326→          <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 gap-2">
   327→            {topics.map((t: TopicScore) => {
   328→              const cls = colorFor(t.accuracy);
   329→              return (
   330→                <div
   331→                  key={`${t.subject}|${t.topic}`}
   332→                  className={cn('rounded-md p-2.5 border border-white/40', cls)}
   333→                  title={`${t.subject} · ${t.topic} · ${t.accuracy}% · ${t.correct}/${Math.max(1, Math.round(t.total / Math.max(1, t.scored || 1)))} correct`}
   334→                >
   335→                  <p className="text-[10px] opacity-80 uppercase tracking-wide truncate">{t.subject}</p>
   336→                  <p className="text-xs font-semibold truncate">{t.topic}</p>
   337→                  <p className="text-sm font-bold mt-0.5">{t.accuracy}%</p>
   338→                </div>
   339→              );
   340→            })}
   341→          </div>
   342→        )}
   343→        <div className="flex items-center justify-end gap-2 mt-3 text-[10px] text-muted-foreground">
   344→          <span>Low</span>
   345→          <span className="h-2.5 w-6 bg-rose-500 rounded" />
   346→          <span className="h-2.5 w-6 bg-amber-500 rounded" />
   347→          <span className="h-2.5 w-6 bg-amber-300 rounded" />
   348→          <span className="h-2.5 w-6 bg-blue-300 rounded" />
   349→          <span className="h-2.5 w-6 bg-blue-500 rounded" />
   350→          <span>High</span>
   351→        </div>
   352→      </CardContent>
   353→    </Card>
   354→  );
   355→}

   358→function BehaviorPanel({ attempt }: { attempt: ExamAttempt }) {
   359→  const b = attempt.behavior;
   360→
   361→  if (!b) {
   362→    return (
   363→      <Card className="border-stone-200">
   364→        <CardContent className="pt-10 pb-10 flex flex-col items-center text-center gap-3">
   365→          <div className="h-12 w-12 rounded-full bg-stone-100 flex items-center justify-center">
   366→            <Brain className="h-6 w-6 text-stone-400" />
   367→          </div>
   368→          <div>
   369→            <h3 className="font-semibold">Behavior analysis not available</h3>
   370→            <p className="text-sm text-muted-foreground mt-1 max-w-md">
   371→              We couldn&apos;t capture detailed timing data for this attempt. Future mocks will include a full behavioural breakdown.
   372→            </p>
   373→          </div>
   374→        </CardContent>
   375→      </Card>
   376→    );
   377→  }
   378→
   379→  const totalTime = attempt.durationSec;
   380→  const speed = attempt.speed;
   381→  const idlePct = totalTime > 0 ? Math.round((b.idleTimeSec / totalTime) * 100) : 0;
   382→  const deciles = b.speedProgression;
   383→  const maxDecileSec = Math.max(1, ...deciles.map((d) => d.avgSec));
   384→  const overallAvg = deciles.length > 0
   385→    ? Math.round(deciles.reduce((a, d) => a + d.avgSec, 0) / deciles.length)
   386→    : 0;
   387→
   388→  const paceStyle =
   389→    b.paceTrend === 'speeding-up'
   390→      ? { bg: 'bg-blue-50 border-blue-200 text-blue-800', icon: TrendingUp, label: 'Speeding up', desc: 'You picked up pace through the paper — momentum built nicely. Watch for rushed answers near the end.' }
   391→      : b.paceTrend === 'slowing-down'
   392→        ? { bg: 'bg-amber-50 border-amber-200 text-amber-800', icon: AlertTriangle, label: 'Slowing down', desc: 'Fatigue may have set in during the second half. Consider endurance drills and shorter focused sessions.' }
   393→        : { bg: 'bg-teal-50 border-teal-200 text-teal-800', icon: Activity, label: 'Steady', desc: 'You maintained a consistent pace throughout. Strong exam discipline.' };
   394→
   395→  const PaceIcon = paceStyle.icon;
   396→
   397→  return (
   398→    <div className="space-y-4">
   399→      {/* Stat cards */}
   400→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   401→        <StatCard label="Total Time" value={fmtDuration(totalTime)} sub={`${attempt.avgTimePerQuestion}s/Q avg`} icon={Clock} accent="emerald" />
   402→        <StatCard label="Speed" value={`${speed}`} sub="questions / hr" icon={Zap} accent="teal" />
   403→        <StatCard label="Idle Time" value={fmtDuration(b.idleTimeSec)} sub={`${b.idlePauses} pauses · ${idlePct}%`} icon={AlertCircle} accent="amber" />
   404→        <StatCard label="Time of Day" value={b.timeOfDay} sub={`started ${String(b.startedAtHour).padStart(2, '0')}:00`} icon={Clock} accent="rose" />
   405→      </div>
   406→
   407→      {/* Pace trend banner */}
   408→      <Card className={cn('border', paceStyle.bg)}>
   409→        <CardContent className="pt-6 flex items-start gap-3">
   410→          <div className="h-9 w-9 rounded-lg bg-white/70 flex items-center justify-center flex-shrink-0">
   411→            <PaceIcon className="h-4 w-4" />
   412→          </div>
   413→          <div>
   414→            <p className="text-xs uppercase tracking-wide opacity-70">Pace Trend</p>
   415→            <p className="font-semibold">{paceStyle.label}</p>
   416→            <p className="text-sm mt-1 leading-relaxed">{paceStyle.desc}</p>
   417→          </div>
   418→        </CardContent>
   419→      </Card>
   420→
   421→      {/* Speed progression bar chart */}
   422→      <Card className="border-stone-200">
   423→        <CardHeader>
   424→          <CardTitle className="text-base flex items-center gap-2">
   425→            <BarChart3 className="h-4 w-4 text-blue-600" /> Speed Progression (deciles)
   426→          </CardTitle>
   427→          <CardDescription>
   428→            Average seconds per question, split into 10 equal buckets across the paper. Green = on-pace, Amber = rushing, Rose = fatigue.
   429→          </CardDescription>
   430→        </CardHeader>
   431→        <CardContent>
   432→          <div className="flex items-end justify-between gap-1.5 h-40">
   433→            {deciles.map((d) => {
   434→              const h = Math.max(4, Math.round((d.avgSec / maxDecileSec) * 100));
   435→              const delta = d.avgSec - overallAvg;
   436→              const cls =
   437→                delta < -overallAvg * 0.2
   438→                  ? 'bg-amber-400'
   439→                  : delta > overallAvg * 0.3
   440→                    ? 'bg-rose-400'
   441→                    : 'bg-blue-500';
   442→              return (
   443→                <div key={d.decile} className="flex-1 flex flex-col items-center gap-1">
   444→                  <span className="text-[10px] text-muted-foreground">{d.avgSec}s</span>
   445→                  <div className="w-full bg-stone-100 rounded-t-md overflow-hidden flex items-end" style={{ height: '120px' }}>
   446→                    <div
   447→                      className={cn('w-full rounded-t-md transition-all', cls)}
   448→                      style={{ height: `${h}%` }}
   449→                      title={`D${d.decile} · ${d.avgSec}s avg`}

   451→                  </div>

   458→            <span>Average: <strong className="text-stone-700">{overallAvg}s/Q</strong></span>

   460→              <span className="flex items-center gap-1"><span className="h-2 w-3 bg-blue-500 rounded-sm" /> On-pace</span>

   469→        {/* Time spent per subject */}

   472→            <CardTitle className="text-base flex items-center gap-2">

   587→function CoachingCard({ b, attempt }: { b: BehaviorAnalysis; attempt: ExamAttempt }) {
   588→  const recs: { tone: 'warn' | 'good' | 'tip'; title: string; body: string }[] = [];
   589→
   590→  if (b.rapidGuesses > 5) {
   591→    recs.push({
   592→      tone: 'warn',
   593→      title: `${b.rapidGuesses} rapid guesses detected`,
   594→      body: 'You answered several questions in under 10 seconds. These are likely blind guesses. Train yourself to flag and skip instead — guessing racks up negative marks.',
   595→    });
   596→  }
   597→  if (b.idlePauses > 3) {
   598→    recs.push({
   599→      tone: 'warn',
   600→      title: `${b.idlePauses} long idle pauses`,
   601→      body: `You spent ${fmtDuration(b.idleTimeSec)} frozen on questions. If a question stumps you for >60s, mark it for review and move on — your brain will often solve it subconsciously while you tackle the next one.`,
   602→    });
   603→  }
   604→  if (b.paceTrend === 'slowing-down') {
   605→    recs.push({
   606→      tone: 'warn',
   607→      title: 'End-of-paper fatigue',
   608→      body: 'Your pace dropped in the second half. Build exam endurance with full-length timed mocks every week and stay hydrated during the test.',
   609→    });
   610→  }
   611→  if (b.paceTrend === 'speeding-up' && b.rapidGuesses > 3) {
   612→    recs.push({
   613→      tone: 'tip',
   614→      title: 'Rein in your finish sprint',
   615→      body: 'You sped up at the end but accuracy often drops with speed. Reserve the last 5 minutes for review, not new attempts.',
   616→    });
   617→  }
   618→  if (b.difficultyTimeGap < 15 && attempt.accuracy < 70) {
   619→    recs.push({
   620→      tone: 'tip',
   621→      title: 'Spend more time on hard questions',
   622→      body: 'You spent almost the same time on easy and hard questions. Hard questions deserve 50-100% more time — but only after you\'ve banked the easy marks.',
   623→    });
   624→  }
   625→
   626→  const isExcellent = recs.length === 0;
   627→
   628→  return (
   629→    <Card className="border-blue-200 bg-blue-50/30">
   630→      <CardHeader>
   631→        <CardTitle className="text-base flex items-center gap-2">
   632→          <Lightbulb className="h-4 w-4 text-blue-600" /> AI Behaviour Coaching
   633→        </CardTitle>
   634→        <CardDescription>Personalised recommendations based on your exam behaviour</CardDescription>
   635→      </CardHeader>
   636→      <CardContent className="space-y-2">
   637→        {isExcellent ? (
   638→          <div className="flex items-start gap-3 p-3 rounded-md bg-blue-100 border border-blue-300">
   639→            <CheckCircle2 className="h-5 w-5 text-blue-700 flex-shrink-0 mt-0.5" />
   640→            <div>
   641→              <p className="font-semibold text-blue-900">Excellent exam discipline!</p>
   642→              <p className="text-sm text-blue-800 mt-1">
   643→                You kept a steady pace, avoided excessive idle time, and didn&apos;t rush. Keep this rhythm — you&apos;re in great shape for the real exam.
   644→              </p>
   645→            </div>
   646→          </div>
   647→        ) : (
   648→          recs.map((r, i) => (
   649→            <div
   650→              key={i}
   651→              className={cn(
   652→                'flex items-start gap-3 p-3 rounded-md border',
   653→                r.tone === 'warn'
   654→                  ? 'bg-amber-50 border-amber-200'
   655→                  : r.tone === 'tip'
   656→                    ? 'bg-teal-50 border-teal-200'
   657→                    : 'bg-blue-50 border-blue-200',
   658→              )}
   659→            >
   660→              {r.tone === 'warn'
   661→                ? <AlertTriangle className="h-5 w-5 text-amber-700 flex-shrink-0 mt-0.5" />
   662→                : r.tone === 'tip'
   663→                  ? <Lightbulb className="h-5 w-5 text-teal-700 flex-shrink-0 mt-0.5" />
   664→                  : <CheckCircle2 className="h-5 w-5 text-blue-700 flex-shrink-0 mt-0.5" />}
   665→              <div>
   666→                <p className="font-semibold text-sm text-stone-900">{r.title}</p>
   667→                <p className="text-sm text-stone-700 mt-0.5 leading-relaxed">{r.body}</p>
   668→              </div>
   669→            </div>
   670→          ))
   671→        )}
   672→      </CardContent>
   673→    </Card>
   674→  );
   675→}
