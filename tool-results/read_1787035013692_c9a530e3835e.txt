   306→function daysToExam(examDate?: string): number {
   307→  if (!examDate) return 0;
   308→  const today = new Date();
   309→  today.setHours(0, 0, 0, 0);
   310→  const target = new Date(examDate);
   311→  target.setHours(0, 0, 0, 0);
   312→  return Math.max(0, Math.ceil((target.getTime() - today.getTime()) / 86400000));
   313→}

   315→function ReadinessRing({ value }: { value: number }) {
   316→  const radius = 32;
   317→  const circ = 2 * Math.PI * radius;
   318→  const offset = circ - (Math.min(100, Math.max(0, value)) / 100) * circ;
   319→  const color = value >= 75 ? '#10b981' : value >= 50 ? '#f59e0b' : '#f43f5e';
   320→  return (
   321→    <div className="relative h-20 w-20 flex-shrink-0">
   322→      <svg className="h-20 w-20 -rotate-90" viewBox="0 0 80 80">
   323→        <defs>
   324→          <linearGradient id="ring-grad" x1="0" y1="0" x2="1" y2="1">
   325→            <stop offset="0%" stopColor="#10b981" />
   326→            <stop offset="60%" stopColor="#0d9488" />
   327→            <stop offset="100%" stopColor="#f59e0b" />

   356→function ScoreTrendCard({ scores }: { scores: { label: string; value: number }[] }) {
   357→  if (scores.length === 0) {
   358→    return (
   359→      <Card className="p-5 border-stone-200 h-full">
   360→        <p className="text-sm font-semibold text-stone-900 mb-2 flex items-center gap-1.5">
   361→          <TrendingUp className="h-4 w-4 text-blue-600" /> Score trend
   362→        </p>
   363→        <p className="text-xs text-muted-foreground">No attempts yet. Take your first mock to see your trend.</p>
   364→      </Card>
   365→    );
   366→  }
   367→
   368→  const max = Math.max(...scores.map((s) => s.value), 1);
   369→  const min = Math.min(...scores.map((s) => s.value), 0);
   370→  const range = Math.max(max - min, 1);
   371→  const w = 240;
   372→  const h = 80;
   373→  const pts = scores.map((s, i) => {
   374→    const x = (i / Math.max(scores.length - 1, 1)) * w;
   375→    const y = h - ((s.value - min) / range) * h;
   376→    return { x, y, ...s };
   377→  });
   378→  const path = pts.map((p, i) => `${i === 0 ? 'M' : 'L'}${p.x},${p.y}`).join(' ');
   379→  const area = `${path} L${w},${h} L0,${h} Z`;
   380→
   381→  return (
   382→    <Card className="p-5 border-stone-200 h-full">
   383→      <div className="flex items-center justify-between mb-3">
   384→        <p className="text-sm font-semibold text-stone-900 flex items-center gap-1.5">
   385→          <TrendingUp className="h-4 w-4 text-blue-600" /> Score trend
   386→        </p>
   387→        <Badge variant="outline" className="bg-blue-50 border-blue-200 text-blue-700">
   388→          <ArrowUpRight className="h-3 w-3" /> {scores.length} attempts
   389→        </Badge>
   390→      </div>
   391→      <div className="flex items-end gap-1">
   392→        <svg viewBox={`0 0 ${w} ${h}`} className="w-full h-20" preserveAspectRatio="none">
   393→          <defs>
   394→            <linearGradient id="area-grad" x1="0" y1="0" x2="0" y2="1">
   395→              <stop offset="0%" stopColor="#10b981" stopOpacity="0.4" />
   396→              <stop offset="100%" stopColor="#10b981" stopOpacity="0" />
   397→            </linearGradient>
   398→          </defs>
   399→          <path d={area} fill="url(#area-grad)" />
   400→          <path d={path} fill="none" stroke="#10b981" strokeWidth="2" strokeLinejoin="round" strokeLinecap="round" />
   401→          {pts.map((p, i) => (
   402→            <circle key={i} cx={p.x} cy={p.y} r="2.5" fill="#047857" />
   403→          ))}
   404→        </svg>
   405→      </div>
   406→      <div className="mt-2 flex justify-between text-[10px] text-muted-foreground">
   407→        {scores.map((s, i) => (
   408→          <span key={i}>{s.label}</span>
   409→        ))}
   410→      </div>
   411→    </Card>
   412→  );
   413→}

   469→              <h1 className="text-2xl sm:text-3xl font-bold tracking-tight text-stone-900">
