   307→function daysToExam(examDate?: string): number {
   308→  if (!examDate) return 0;
   309→  const today = new Date();
   310→  today.setHours(0, 0, 0, 0);
   311→  const target = new Date(examDate);
   312→  target.setHours(0, 0, 0, 0);
   313→  return Math.max(0, Math.ceil((target.getTime() - today.getTime()) / 86400000));
   314→}

   316→function ReadinessRing({ value }: { value: number }) {
   317→  const radius = 32;
   318→  const circ = 2 * Math.PI * radius;
   319→  const offset = circ - (Math.min(100, Math.max(0, value)) / 100) * circ;
   320→  const color = value >= 75 ? '#10b981' : value >= 50 ? '#f59e0b' : '#f43f5e';
   321→  return (
   322→    <div className="relative h-20 w-20 flex-shrink-0">
   323→      <svg className="h-20 w-20 -rotate-90" viewBox="0 0 80 80">
   324→        <defs>
   325→          <linearGradient id="ring-grad" x1="0" y1="0" x2="1" y2="1">
   326→            <stop offset="0%" stopColor="#10b981" />
   327→            <stop offset="60%" stopColor="#0d9488" />

   357→function ScoreTrendCard({ scores }: { scores: { label: string; value: number }[] }) {
   358→  if (scores.length === 0) {
   359→    return (
   360→      <Card className="p-5 border-stone-200 h-full">
   361→        <p className="text-sm font-semibold text-stone-900 mb-2 flex items-center gap-1.5">
   362→          <TrendingUp className="h-4 w-4 text-blue-600" /> Score trend
   363→        </p>
   364→        <p className="text-xs text-muted-foreground">No attempts yet. Take your first mock to see your trend.</p>
   365→      </Card>
   366→    );
   367→  }
   368→
   369→  const max = Math.max(...scores.map((s) => s.value), 1);
   370→  const min = Math.min(...scores.map((s) => s.value), 0);
   371→  const range = Math.max(max - min, 1);
   372→  const w = 240;
   373→  const h = 80;
   374→  const pts = scores.map((s, i) => {
   375→    const x = (i / Math.max(scores.length - 1, 1)) * w;
   376→    const y = h - ((s.value - min) / range) * h;
   377→    return { x, y, ...s };
   378→  });
   379→  const path = pts.map((p, i) => `${i === 0 ? 'M' : 'L'}${p.x},${p.y}`).join(' ');
   380→  const area = `${path} L${w},${h} L0,${h} Z`;
   381→
   382→  return (
   383→    <Card className="p-5 border-stone-200 h-full">
   384→      <div className="flex items-center justify-between mb-3">
   385→        <p className="text-sm font-semibold text-stone-900 flex items-center gap-1.5">
   386→          <TrendingUp className="h-4 w-4 text-blue-600" /> Score trend
   387→        </p>
   388→        <Badge variant="outline" className="bg-blue-50 border-blue-200 text-blue-700">
   389→          <ArrowUpRight className="h-3 w-3" /> {scores.length} attempts
   390→        </Badge>
   391→      </div>
   392→      <div className="flex items-end gap-1">
   393→        <svg viewBox={`0 0 ${w} ${h}`} className="w-full h-20" preserveAspectRatio="none">
   394→          <defs>
   395→            <linearGradient id="area-grad" x1="0" y1="0" x2="0" y2="1">
   396→              <stop offset="0%" stopColor="#10b981" stopOpacity="0.4" />
   397→              <stop offset="100%" stopColor="#10b981" stopOpacity="0" />
   398→            </linearGradient>
   399→          </defs>
   400→          <path d={area} fill="url(#area-grad)" />
   401→          <path d={path} fill="none" stroke="#10b981" strokeWidth="2" strokeLinejoin="round" strokeLinecap="round" />
   402→          {pts.map((p, i) => (
   403→            <circle key={i} cx={p.x} cy={p.y} r="2.5" fill="#047857" />
   404→          ))}
   405→        </svg>
   406→      </div>
   407→      <div className="mt-2 flex justify-between text-[10px] text-muted-foreground">
   408→        {scores.map((s, i) => (
   409→          <span key={i}>{s.label}</span>
   410→        ))}
   411→      </div>
   412→    </Card>
   413→  );
   414→}
